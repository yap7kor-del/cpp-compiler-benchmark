cmake_minimum_required(VERSION 3.15)

project(EmbeddedApp CXX)

# Force C++20 standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 1. COMPILER-SPECIFIC CONFIGURATIONS (GCC & Clang)
# Here we check which compiler CMake has detected (via Conan Toolchain or System)
# and apply strict compiler warning flags accordingly.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(STATUS "Configuring for GCC compiler...")
    add_compile_options(
        -Wall 
        -Wextra 
        -Wpedantic 
        -Wdouble-promotion       # Warn if float is implicitly promoted to double (crucial for embedded)
        -fdiagnostics-color=always
    )
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(STATUS "Configuring for Clang compiler...")
    add_compile_options(
        -Wall 
        -Wextra 
        -Wpedantic 
        -Wconversion            # Warn on implicit type conversions
        -fcolor-diagnostics
    )
else()
    message(STATUS "Configuring for Compiler: ${CMAKE_CXX_COMPILER_ID}")
endif()

# 2. LOCATE DEPENDENCIES (Generated by Conan CMakeDeps)
# This looks for the "fmt-config.cmake" files generated by Conan inside your build directory
find_package(fmt REQUIRED)

# Support both compiled fmt (fmt::fmt) and header-only fmt (fmt::fmt-header-only)
if(TARGET fmt::fmt-header-only AND NOT TARGET fmt::fmt)
    add_library(fmt::fmt ALIAS fmt::fmt-header-only)
endif()

# 3. DEFINE TARGET & LINK DEPENDENCIES
# Create our executable target from source files and sub-folders
add_executable(EmbeddedApp 
    src/main.cpp
    src/math/matrix.hpp
    src/math/matrix.cpp
    src/dsp/filter.hpp
    src/dsp/filter.cpp
)

# Set include directories so headers can be found relative to src/
target_include_directories(EmbeddedApp PRIVATE src)

# Link the fmt library target (defined in Conan's fmtTargets.cmake)
target_link_libraries(EmbeddedApp PRIVATE fmt::fmt)

# 4. ARM BARE-METAL & QEMU SEMIHOSTING SPECIFICATIONS
if(CMAKE_CXX_COMPILER MATCHES "arm-none-eabi-")
    message(STATUS "Configuring bare-metal ARM linker specifications for QEMU semihosting...")
    target_link_options(EmbeddedApp PRIVATE
        "-mcpu=cortex-m4"
        "-mthumb"
        "--specs=rdimon.specs"   # Semihosting support for QEMU I/O (printf, puts, exit)
        "--specs=nosys.specs"    # Provides default stubs for non-I/O OS calls
        "-Wl,--defsym=_getentropy=0" # Resolves missing Newlib 4.4 entropy symbol
    )

    # Link the semihosting runtime library
    target_link_libraries(EmbeddedApp PRIVATE rdimon)
endif()
