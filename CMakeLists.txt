cmake_minimum_required(VERSION 3.15)
project(EmbeddedApp CXX)

# Force C++20 standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 1. COMPILER-SPECIFIC CONFIGURATIONS (GCC & Clang)
# Here we check which compiler CMake has detected (via Conan Toolchain or System)
# and apply strict compiler warning flags accordingly.

if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(STATUS "Configuring for GCC compiler...")
    add_compile_options(
        -Wall 
        -Wextra 
        -Wpedantic 
        -Wdouble-promotion       # Warn if float is implicitly promoted to double (crucial for embedded)
        -fdiagnostics-color=always
    )
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(STATUS "Configuring for Clang compiler...")
    add_compile_options(
        -Wall 
        -Wextra 
        -Wpedantic 
        -Wconversion            # Warn on implicit type conversions
        -fcolor-diagnostics
    )
else()
    message(STATUS "Configuring for Compiler: ${CMAKE_CXX_COMPILER_ID}")
endif()

# 2. LOCATE DEPENDENCIES (Generated by Conan CMakeDeps)
# This looks for the "fmt-config.cmake" files generated by Conan inside your build directory
find_package(fmt REQUIRED)

# 3. DEFINE TARGET & LINK DEPENDENCIES
# Create our executable target from source files and sub-folders
add_executable(EmbeddedApp 
    src/main.cpp
    src/math/matrix.hpp
    src/math/matrix.cpp
    src/dsp/filter.hpp
    src/dsp/filter.cpp
)

# Set include directories so headers can be found relative to src/
target_include_directories(EmbeddedApp PRIVATE src)

# Link the fmt library target (defined in Conan's fmtTargets.cmake)
target_link_libraries(EmbeddedApp PRIVATE fmt::fmt)
